  async function playSequence(seq,{whammy=true}={}){
    const token=++sceneToken;sceneBusy=true;$('#centerScreen').classList.add('playing');$('#spinBtn').disabled=true;
    if(whammy)whammySting();
    for(const [name,duration,motion,text,fx,doFlash] of seq.frames){
      if(token!==sceneToken)return;
      setMascot(name,motion);caption(text,duration);if(fx)showFx(fx);if(doFlash)flash();sceneSound(seq.sound);
      await sleep(duration);
    }
    if(token!==sceneToken)return;
    showFx(null);$('#centerScreen').classList.remove('playing');setMascot('idle','pose-pop');sceneBusy=false;$('#spinBtn').disabled=false;
  }
